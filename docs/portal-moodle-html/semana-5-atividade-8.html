Implementa Pessoa, mixin Ano, Aluno, Professor, Disciplina, Turma e Historico (com validação de notas finitas 0..10, cálculo de média da lista de notas do aluno e aprovação media > 6):

import 'package:flutter_test/flutter_test.dart';

abstract class Pessoa {
  late int _id;
  final String nome;
  Pessoa(this.nome);
  int get id => _id;
  set id(int id) {
    if (id <= 0) throw ArgumentError('Identificador deve ser positivo.');
    _id = id;
  }
}

mixin Ano {
  late int _ano;
  int get ano => _ano;
  set ano(int ano) {
    if (ano <= 0) throw ArgumentError('Ano deve ser positivo.');
    _ano = ano;
  }
}

class Aluno extends Pessoa with Ano {
  Aluno(super.nome, int ano) { this.ano = ano; }
}

class Professor extends Pessoa { Professor(super.nome); }

class Disciplina { final String nome; Disciplina(this.nome); }

class Turma with Ano {
  final Disciplina disciplina;
  final Professor professor;
  final List<Aluno> _alunos = [];

  Turma(this.disciplina, this.professor, int ano) { this.ano = ano; }

  void matricular(Aluno aluno) {
    if (aluno.ano != ano) {
      throw ArgumentError('O ano do aluno deve ser igual ao ano da turma.');
    }
    if (_alunos.contains(aluno)) {
      throw ArgumentError('Aluno já está matriculado nesta turma.');
    }
    _alunos.add(aluno);
  }
}

class Historico extends Turma {
  final Map<Aluno, List<double>> notas = {};
  Historico(super.disciplina, super.professor, super.ano);

  @override
  void matricular(Aluno aluno) {
    super.matricular(aluno);
    notas[aluno] = [];
  }

  void adicionarNota(Aluno aluno, double nota) {
    if (!notas.containsKey(aluno)) {
      throw ArgumentError('Aluno não está matriculado nesta turma.');
    }
    if (!nota.isFinite || nota < 0 || nota > 10) {
      throw ArgumentError(
        'A nota deve ser finita e estar entre 0 e 10.',
      );
    }
    notas[aluno]!.add(nota);
  }

  double media(Aluno aluno) {
    final listaNotas = notas[aluno];
    if (listaNotas == null || listaNotas.isEmpty) return 0.0;
    return listaNotas.reduce((soma, nota) => soma + nota) / listaNotas.length;
  }

  bool isAprovado(Aluno aluno) => media(aluno) > 6;
}

void main() {
  group('Atividade 8 — Semana 5: Orientação a Objetos', () {
    late Historico historico;
    setUp(() {
      historico = Historico(Disciplina('Flutter'), Professor('Edilson Luiz Nascimento'), 2026);
    });

    test('matricula aluno e registra identificador', () {
      final aluno = Aluno('Maria', 2026)..id = 1;
      historico.matricular(aluno);
      expect(aluno.id, 1);
      expect(historico.media(aluno), 0.0);
      expect(historico.isAprovado(aluno), false);
    });
    test('rejeita identificador inválido', () {
      final aluno = Aluno('Paula', 2026);
      expect(() => aluno.id = 0, throwsA(isA<ArgumentError>()));
    });
    test('rejeita ano inválido', () {
      expect(() => Aluno('Pedro', 0), throwsA(isA<ArgumentError>()));
    });
    test('rejeita matrícula em turma de outro ano', () {
      expect(() => historico.matricular(Aluno('Ana', 2025)), throwsA(isA<ArgumentError>()));
    });
    test('rejeita matrícula duplicada', () {
      final aluno = Aluno('Ana', 2026);
      historico.matricular(aluno);
      expect(() => historico.matricular(aluno), throwsA(isA<ArgumentError>()));
    });
    test('calcula média das notas', () {
      final aluno = Aluno('Maria', 2026);
      historico.matricular(aluno);
      historico.adicionarNota(aluno, 8.0);
      historico.adicionarNota(aluno, 7.0);
      expect(historico.media(aluno), 7.5);
    });
    test('aprova média acima de seis', () {
      final aluno = Aluno('Maria', 2026);
      historico.matricular(aluno);
      historico.adicionarNota(aluno, 8.0);
      historico.adicionarNota(aluno, 7.0);
      expect(historico.isAprovado(aluno), true);
    });
    test('não aprova média igual a seis', () {
      final aluno = Aluno('Paula', 2026);
      historico.matricular(aluno);
      historico.adicionarNota(aluno, 6.0);
      historico.adicionarNota(aluno, 6.0);
      expect(historico.isAprovado(aluno), false);
    });
    test('rejeita nota fora do intervalo de zero a dez', () {
      final aluno = Aluno('Carlos', 2026);
      historico.matricular(aluno);
      expect(() => historico.adicionarNota(aluno, -1), throwsA(isA<ArgumentError>()));
      expect(() => historico.adicionarNota(aluno, 11), throwsA(isA<ArgumentError>()));
    });
    test('rejeita nota para aluno não matriculado', () {
      expect(() => historico.adicionarNota(Aluno('Rita', 2026), 8), throwsA(isA<ArgumentError>()));
    });
    test('rejeita valores especiais como NaN, infinito positivo e infinito negativo', () {
      final aluno = Aluno('Carlos', 2026);
      historico.matricular(aluno);
      expect(() => historico.adicionarNota(aluno, double.nan), throwsA(isA<ArgumentError>()));
      expect(() => historico.adicionarNota(aluno, double.infinity), throwsA(isA<ArgumentError>()));
      expect(() => historico.adicionarNota(aluno, double.negativeInfinity), throwsA(isA<ArgumentError>()));
    });
  });
}
